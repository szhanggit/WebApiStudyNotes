IF EXISTS ( SELECT  1
            FROM    dbo.sysobjects
            WHERE   id = OBJECT_ID(N'[dbo].[spEditEvents]')
                    AND OBJECTPROPERTY(id, N'IsProcedure') = 1 )
    DROP PROCEDURE [dbo].[spEditEvents]
GO

CREATE PROCEDURE [dbo].[spEditEvents]
	@ErrorMessage NVARCHAR(4000) OUTPUT,
	@Events Event_Input Readonly
AS
BEGIN
	SET NOCOUNT ON;
	
    BEGIN TRAN
    BEGIN TRY	
		declare @count int;
		select @count = count(1) from @Events	

		if(@count > 0)
		begin
			update e
			set e.EventName = ev.EventName
			, e.StartDateTime = ev.StartDateTime
			, e.EndDateTime = ev.EndDateTime
			, e.Fee = ev.Fee
			, e.[Status] = ev.[Status]
			from Event e with(nolock) 
			inner join @Events ev on e.Id = ev.Id
		end
		
    END TRY
    BEGIN CATCH
        SELECT @ErrorMessage = ERROR_MESSAGE();
        ROLLBACK TRANSACTION;
		return 0;
    END CATCH;
	IF @@TRANCOUNT > 0
	begin
		COMMIT TRANSACTION;
		return 1;
	end
END
GO

CREATE PROCEDURE [dbo].[spDeleteEvents]
	@ErrorMessage NVARCHAR(4000) OUTPUT,
	@Events EventIdList_Input Readonly
AS
BEGIN
	SET NOCOUNT ON;
	
    BEGIN TRAN
    BEGIN TRY	
		declare @count int;
		select @count = count(1) from @Events	

		if(@count > 0)
		begin
			delete e from Event e with(nolock) 
			inner join @Events ev on e.Id = ev.Id
		end
		
    END TRY
    BEGIN CATCH
        SELECT @ErrorMessage = ERROR_MESSAGE();
        ROLLBACK TRANSACTION;
		return 0;
    END CATCH;
	IF @@TRANCOUNT > 0
	begin
		COMMIT TRANSACTION;
		return 1;
	end
END
GO

CREATE PROCEDURE [dbo].[spAddEvents]
	@ErrorMessage NVARCHAR(4000) OUTPUT,
	@Events Event_Input Readonly
AS
BEGIN
	SET NOCOUNT ON;
	
    BEGIN TRAN
    BEGIN TRY	
		declare @count int;
		select @count = count(1) from @Events	

		if(@count > 0)
		begin
			insert into Event (EventName, StartDateTime, EndDateTime, Fee, [Status])
			select EventName, StartDateTime, EndDateTime, Fee, [Status] from @Events;			
		end
		
    END TRY
    BEGIN CATCH
        SELECT @ErrorMessage = ERROR_MESSAGE();
        ROLLBACK TRANSACTION;
		return 0;
    END CATCH;
	IF @@TRANCOUNT > 0
	begin
		COMMIT TRANSACTION;
		return 1;
	end
END
GO
